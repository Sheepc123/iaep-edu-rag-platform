### 3. 核心代码框架

#### FastAPI应用入口 (app/main.py)
```python
from fastapi import FastAPI, Middleware
from fastapi.middleware.cors import CORSMiddleware
from fastapi.middleware.trustedhost import TrustedHostMiddleware
from contextlib import asynccontextmanager

from app.core.config import settings
from app.core.database import engine
from app.api.v1.router import api_router

@asynccontextmanager
async def lifespan(app: FastAPI):
    # 启动时执行
    print("🚀 启动智能教育平台后端服务")
    yield
    # 关闭时执行
    print("🛑 关闭智能教育平台后端服务")

app = FastAPI(
    title="智能教育实训平台API",
    description="基于AI的智能教育实训平台后端服务",
    version="1.0.0",
    lifespan=lifespan,
    docs_url="/docs",
    redoc_url="/redoc"
)

# 中间件配置
app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.ALLOWED_HOSTS,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 路由注册
app.include_router(api_router, prefix="/api/v1")

@app.get("/")
async def root():
    return {"message": "智能教育实训平台API服务正在运行"}

@app.get("/health")
async def health_check():
    return {"status": "healthy", "version": "1.0.0"}
```

### 4. AI服务集成

#### AI服务类 (app/services/ai_service.py)
```python
import openai
from typing import Dict, List, Optional
from app.core.config import settings

class AIEducationService:
    def __init__(self):
        self.client = openai.AsyncOpenAI(
            api_key=settings.DEEPSEEK_API_KEY,
            base_url=settings.DEEPSEEK_BASE_URL
        )
        
    async def generate_response(
        self, 
        user_input: str, 
        context: Optional[Dict] = None,
        conversation_history: Optional[List[Dict]] = None
    ) -> str:
        """生成AI回复"""
        try:
            messages = [
                {
                    "role": "system",
                    "content": self._build_system_prompt(context)
                }
            ]
            
            # 添加对话历史
            if conversation_history:
                messages.extend(conversation_history[-10:])  # 保留最近10条
            
            messages.append({
                "role": "user",
                "content": user_input
            })
            
            response = await self.client.chat.completions.create(
                model="deepseek-chat",
                messages=messages,
                max_tokens=1000,
                temperature=0.7
            )
            
            return response.choices[0].message.content
            
        except Exception as e:
            logger.error(f"AI生成回复失败: {e}")
            return "抱歉，AI助手暂时无法回复，请稍后再试。"
    
    def _build_system_prompt(self, context: Optional[Dict] = None) -> str:
        """构建系统提示词"""
        base_prompt = """
        你是一个专业的AI学习助手，专门帮助学生学习和解答问题。
        
        你的特点：
        1. 友好、耐心、专业
        2. 善于解释复杂概念
        3. 能够提供学习建议和方法
        4. 会推荐相关的练习和资源
        
        请根据学生的问题提供有帮助的回答。
        """
        
        if context:
            context_info = f"\n\n当前学习上下文：{context}"
            base_prompt += context_info
            
        return base_prompt
```

```dockerfile
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

EXPOSE 8000

CMD ["uvicorn", "app.main:app", "--host", "0.0.0.0", "--port", "8000"]
```

### Docker Compose
```yaml

version: '3.8'

services:
  api:
    build: .

      - "8000:8000"
    environment:
      - DATABASE_URL=postgresql://edu_user:password@db:5432/education_platform

      - db
```
